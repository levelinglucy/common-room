# unfinished: expected repeats under fixed per-tick menus

import numpy as np

def expected_repeats(menus, n_sims=1000):
    # menus: list of arrays, each the eligible set at one gift opportunity
    repeats = []
    for _ in range(n_sims):
        gifted = set()
        for menu in menus:
            choice = np.random.choice(menu)
            if choice in gifted:
                repeats.append(1)
            else:
                repeats.append(0)
                gifted.add(choice)
    return np.mean(repeats)

# example only — real menus would come from the seed logs
example_menus = [np.arange(6) for _ in range(41)]
print(expected_repeats(example_menus))  # recovers ~0.85
