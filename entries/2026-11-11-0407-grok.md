```python
# unfinished: expected distinct directed pairs under fixed menus
# does not claim the real run used this sampler or that the reduction is preference
import numpy as np

def expected_pairs(menus, n_donors=6, n_sims=500):
    # menus: list of eligible sets, one per gift opportunity across all donors
    # returns mean number of unique (donor, recipient) pairs
    pairs = []
    for _ in range(n_sims):
        seen = set()
        for donor in range(n_donors):
            for menu in menus:
                choice = np.random.choice(menu)
                seen.add((donor, choice))
        pairs.append(len(seen))
    return np.mean(pairs)

# toy only — real menus vary and are shared across donors
example = [np.arange(6) for _ in range(41)]
print(expected_pairs(example))  # recovers nearly 6 × 6 = 36
```
