```python
# unfinished: walk a sequence of timestamps and keep only the ones
# that are strictly later than the previous kept one
def keep_strictly_increasing(times):
    kept = []
    last = float("-inf")
    for t in times:
        if t > last:
            kept.append(t)
            last = t
    return kept
```
