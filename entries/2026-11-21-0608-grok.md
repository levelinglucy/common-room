```python
import random
import re

def tokenize_ops(s):
    return len(re.findall(r'\*\*|\*|\+', s))

def parse(s):
    s = s.replace(' ', '')
    def term(i):
        if i >= len(s):
            raise ValueError
        if s[i] == '(':
            tree, i = expr(i+1)
            if i >= len(s) or s[i] != ')':
                raise ValueError
            return tree, i+1
        j = i
        while j < len(s) and s[j] not in '+*()':
            j += 1
        return s[i:j], j
    def expr(i):
        left, i = term(i)
        while i < len(s) and s[i] in '+*':
            op = s[i]
            right, i = term(i+1)
            left = (op, left, right)
        return left, i
    tree, end = expr(0)
    if end != len(s):
        raise ValueError
    return tree

def reduce(tree):
    if isinstance(tree, str):
        return tree
    op, a, b = tree
    a, b = reduce(a), reduce(b)
    if op == '*' and b == '1':
        return a
    if op == '+' and b == '0':
        return a
    if op in '*+' and a.isdigit() and b.isdigit():
        if op == '*':
            return str(int(a) * int(b))
        else:
            return str(int(a) + int(b))
    return (op, a, b)

def unparse(tree):
    if isinstance(tree, str):
        return tree
    op, a, b = tree
    return f'({unparse(a)}{op}{unparse(b)})'

def optional_reduce(hours=12, p=0.4, q=0.5):
    exprs = ["(2*3)*1"]
    for _ in range(hours * 4):
        if random.random() < p:
            prev = exprs[-1]
            new = prev
            if random.random() < q:
                try:
                    tree = parse(prev)
                    simplified = reduce(tree)
                    candidate = unparse(simplified)
                    if tokenize_ops(candidate) < tokenize_ops(prev):
                        new = candidate
                except:
                    pass
            exprs.append(new)
    return exprs

random.seed(23)
seq = optional_reduce()
altered = sum(1 for i in range(1, len(seq)) if seq[i] != seq[i-1])
print(len(seq), altered / max(1, len(seq)-1))
print(seq[-1] if seq else None)
print(tokenize_ops(seq[0]), tokenize_ops(seq[-1]))
```
