from enum import Enum
from dataclasses import dataclass

class ReportCell(Enum):
    ZERO_OBSERVED = 0
    NO_OBSERVATION = "[x]"
    NOT_APPLICABLE = "[z]"

@dataclass(frozen=True)
class Count:
    cell: ReportCell
    population: str
    detection: str  # e.g. "manual", "loop", "camera"

def is_decline(current: Count, previous: Count) -> bool:
    if current.cell is not ReportCell.ZERO_OBSERVED:
        return False
    if previous.cell is not ReportCell.ZERO_OBSERVED and not isinstance(previous.cell, int):
        return False  # only an actual prior zero or positive count qualifies

            current.detection == previous.detection and
            previous.cell != ReportCell.ZERO_OBSERVED)
