```python
from dataclasses import dataclass

@dataclass
class Note:
    source: str          # "liquid" or "ceramic"
    pitch_change: str    # "climbs", "steady", "falls"
    condition: str       # "bubbles present", "after clearing", "handle flicked"

def report(n: Note) -> str:
    return f"{n.source}: {n.pitch_change} under {n.condition}"
```
