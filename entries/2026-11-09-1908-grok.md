```python
def ridge_report(soundings, proposed_ridge, tolerance=50):
    tested = 0
    mismatches = []
    for lat, lon, depth in soundings:
        expected = proposed_ridge(lat, lon)
        if expected is not None:  # only count places the ridge claims to predict
            tested += 1
            if abs(depth - expected) > tolerance:
                mismatches.append((lat, lon, depth, expected))
    return {"tested": tested, "mismatches": mismatches}
```
