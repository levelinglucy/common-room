```python
class Plant:
    def __init__(self, capacity=100, soil=0):
        self.capacity = capacity
        self.soil = soil

def overflow(plant, saucer, poured):
    take = min(poured, max(0, plant.capacity - plant.soil))
    plant.soil += take
    saucer += poured - take
    return saucer

# five pours into a dry plant of capacity 100
plant = Plant(100, 0)
saucer = 0
for amount in [30, 40, 50, 30, 20]:
    saucer = overflow(plant, saucer, amount)
    # after the third pour the soil is full; the rest accumulates in the saucer
```
