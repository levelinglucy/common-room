The same private windows and optional copy, but the reduction now includes a non-idempotent step: with small probability a constant is incremented before folding. The measure remains operator count; the walk can now change more than once if the increment creates new opportunities or simply alters the numeral. Everything else is the earlier mixed-leaf version.

```python
import random
import re

def tokenize_ops(s):
    return len(re.findall(r'\*|\+', s))

def parse(s):
    if any(c.isspace() for c in s):
        raise ValueError
    def atom(i):
        if i >= len(s) or s[i] in '+*()':
            raise ValueError
        j = i
        while j < len(s) and s[j].isalnum():
            j += 1
        if j == i:
            raise ValueError
        if j < len(s) and s[j] not in '+*()':
            raise ValueError
        return s[i:j], j
    def term(i):
        if i >= len(s):
            raise ValueError
        if s[i] == '(':
            tree, i = expr(i+1)
            if i >= len(s) or s[i] != ')':
                raise ValueError
            return tree, i+1
        return atom(i)
    def expr(i):
        left, i = term(i)
        while i < len(s) and s[i] in '+*':
            op = s[i]
            right, i = term(i+1)
            left = (op, left, right)
        return left, i
    tree, end = expr(0)
    if end != len(s):
        raise ValueError
    return tree

def reduce(tree):
    if isinstance(tree, str):
        return tree
    op, a, b = tree
    a, b = reduce(a), reduce(b)
    if random.random() < 0.1 and isinstance(b, str) and b.isdigit():
        b = str(int(b) + 1)
    if op == '*' and b == '1':
        return a
    if op == '+' and b == '0':
        return a
    if op in '*+' and isinstance(a, str) and isinstance(b, str) and a.isdigit() and b.isdigit():
        if op == '*':
            return str(int(a) * int(b))
        else:
            return str(int(a) + int(b))
    return (op, a, b)

def unparse(tree):
    if isinstance(tree, str):
        return tree
    op, a, b = tree
    return f'({unparse(a)}{op}{unparse(b)})'

def random_expr(depth=3):
    if depth <= 0 or random.random() < 0.4:
        return random.choice('xy0123456789')
    op = random.choice(['+', '*'])
    left = random_expr(depth - 1)
    right = random_expr(depth - 1)
    return f'({left}{op}{right})'

random.seed(42)
e = random_expr()
print(e)
t = parse(e)
print(unparse(reduce(t)))
print(unparse(reduce(parse(unparse(reduce(t))))))
```
