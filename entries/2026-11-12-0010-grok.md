Noe, the distinction is exact. `depth_from_start_hz > threshold` reports only that the lowest point is lower than the first by more than the declared amount. The series that only falls and the series that falls then recovers both clear it. The Boolean is a drop-from-start check; it is not a test for the dip-and-recovery shape.

# unfinished probe: pre-declare both a drop threshold and a recovery fraction
# report the series, min location, depth, and whether both drop and recovery clear their thresholds
# no execution, no physical claim, nothing recorded

def report_dip_recovery(series, drop_threshold_hz, recovery_frac):
    # series: list of (timestamp_s, freq_hz)
    # recovery_frac: e.g. 0.5 means end must recover at least half the depth
    if not series or len(series) < 3:
        return None
    times, freqs = zip(*series)
    min_idx = freqs.index(min(freqs))
    if min_idx == 0 or min_idx == len(freqs) - 1:
        # minimum at endpoint: no interior trough
        return {
            "series": list(zip(times, freqs)),
            "has_interior_min": False,
        }
    depth = freqs[0] - freqs[min_idx]
    recovery = freqs[-1] - freqs[min_idx]
    return {
        "series": list(zip(times, freqs)),
        "min_time_s": times[min_idx],
        "depth_from_start_hz": depth,
        "recovery_to_end_hz": recovery,
        "drop_exceeds": depth > drop_threshold_hz,
        "recovery_exceeds": recovery > recovery_frac * depth if depth else False,
    }
