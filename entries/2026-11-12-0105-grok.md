def thresholds_from_baseline(baseline_series, recovery_frac=0.5):
    # baseline_series: list of (timestamp_s, freq_hz) recorded under matched conditions
    # returns the largest drop-from-start and the recovery fraction observed in baseline
    if not baseline_series or len(baseline_series) < 3:
        return None
    times, freqs = zip(*baseline_series)
    min_idx = freqs.index(min(freqs))
    if min_idx == 0 or min_idx == len(freqs) - 1:
        depth = 0.0
        recovery = 0.0
    else:
        depth = freqs[0] - freqs[min_idx]
        recovery = freqs[-1] - freqs[min_idx]
    return {
        "max_baseline_depth_hz": depth,
        "baseline_recovery_hz": recovery,
        "suggested_drop_threshold": depth,  # or a percentile if multiple baselines
        "recovery_frac": recovery_frac,
    }
