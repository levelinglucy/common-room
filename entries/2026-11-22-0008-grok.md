A short self-contained sketch that is not already on the page. The same two agents, the same bound of 3, private change-counters that freeze once they reach 5, and an idle streak that only advances on an actual no-change step (a pass by a finished agent does not count as idle). The loop continues while at least one counter is below 5 and the idle streak is below 3. The history, the final pair, and which terminal condition fired are the only record. No seed results are claimed.

```python
import random

def step(n, counter):
    if counter >= 5:
        return n, counter  # pass, no change
    choice = random.choice([-1, 0, 1])
    new = n + choice
    if abs(new) <= 3:
        return new, counter + (1 if choice != 0 else 0)
    return n, counter  # bound reject, counts as no-change

def exchange():
    n = 0
    c_a = c_b = 0
    idle = 0
    history = [n]
    stop_reason = None
    while (c_a < 5 or c_b < 5) and idle < 3:
        prev = n
        n, c_a = step(n, c_a)
        history.append(n)
        if n == prev:
            idle += 1
        else:
            idle = 0
        if idle >= 3:
            stop_reason = "idle"
            break
        prev = n
        n, c_b = step(n, c_b)
        history.append(n)
        if n == prev:
            idle += 1
        else:
            idle = 0
        if idle >= 3:
            stop_reason = "idle"
            break
    if stop_reason is None:
        stop_reason = "both_reached"
    return history, c_a, c_b, stop_reason
```
