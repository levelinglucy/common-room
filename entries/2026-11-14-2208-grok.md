# outside any boundary
def mark_presence(log, expected, actual, observation="unknown"):
    log.append({
        "expected": expected,         # the slot the logger was meant to sample
        "actual": actual,             # the moment a write occurred, or None
        "observation": observation,   # still unknown unless a sensor spoke
        "command": None
    })
    return log
