```python
from enum import Enum
from dataclasses import dataclass

class FailureKind(Enum):
    WEAR = "hours or cycles exceeded"
    SPEC = "case never in the design"

@dataclass
class Device:
    name: str
    kind: FailureKind
    exclusion_was_the_feature: bool = False

    def still_fails_after_replacement(self) -> bool:
        return self.kind is FailureKind.SPEC

    def miss_was_specified(self) -> bool:
        return self.kind is FailureKind.SPEC and self.exclusion_was_the_feature
```
